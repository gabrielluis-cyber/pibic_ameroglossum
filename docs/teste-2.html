# Dados em CSV e troca de "," para "."

dados_ameroglossum <- read.csv("ameroglossum.csv")
dados_ameroglossum$long <- as.numeric(gsub(",", ".", dados_ameroglossum$long))
dados_ameroglossum$lat  <- as.numeric(gsub(",", ".", dados_ameroglossum$lat))

# Definindo variáveis (raster's definidos)

bio1 <- rast("wc2.1_10m_bio_1.tif")
bio4 <- rast("wc2.1_10m_bio_4.tif")
bio7 <- rast("wc2.1_10m_bio_7.tif")
bio12 <- rast("wc2.1_10m_bio_12.tif")
elev <- rast("wc2.1_10m_elev.tif")

#Psudoausências

PA.nb.rep = 3
PA.nb.absences = 3000
PA.strategy = "random"

# Resp.var e Resp.xy (Objetos de presença e localidade)

resp.var <- rep(1,nrow(dados_ameroglossum))
resp.xy <- dados_ameroglossum[,c("long","lat")]

# Concatenando variáveis

variaveis <- c(bio1, bio4, bio7, bio12, elev)

# Criando objeto de teste

testebio24 <- BIOMOD_FormatingData(
  resp.var = resp.var,
  expl.var = variaveis,
  resp.xy = resp.xy,
  resp.name = "A.Pernambucense teste 2",
  PA.nb.rep = 3,
  PA.nb.absences = 3000,
  PA.strategy = "random",
)

#Psudoausências

PA.nb.rep = 3
PA.nb.absences = 3000
PA.strategy = "random"

# Resp.var e Resp.xy (Objetos de presença e localidade)

resp.var <- rep(1,nrow(dados_ameroglossum))
resp.xy <- dados_ameroglossum[,c("long","lat")]

# Concatenando variáveis

variaveis <- c(bio1, bio4, bio7, bio12, elev)

# Criando objeto de teste

testebio24 <- BIOMOD_FormatingData(
  resp.var = resp.var,
  expl.var = variaveis,
  resp.xy = resp.xy,
  resp.name = "A.Pernambucense teste 2",
  PA.nb.rep = 3,
  PA.nb.absences = 3000,
  PA.strategy = "random",
)

# Modelando com GLM, GAM, RF e MaxEnt
testebio24_glm <- BIOMOD_Modeling(
  bm.format=testebio24,
  modeling.id = "teste 1, 4, 7, 12, elev",
  models = c("GLM", "GAM", "RF", "MAXNET"),
  CV.strategy = "random",
  CV.nb.rep = 3,
  CV.perc = 0.75,
  metric.eval = c ("TSS", "AUCroc", "AUCprg"),
  var.import = 10
)


# Vendo os resultados
get_evaluations(testebio24_glm)
